cmake_minimum_required(VERSION 3.16)
project(feel_firmware_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(feel_controller src/controller.cpp src/commands.cpp)
target_include_directories(feel_controller PUBLIC include)
target_compile_options(feel_controller PRIVATE -Wall -Wextra -Wpedantic)
add_executable(feel-sim host/simulator.cpp)
target_link_libraries(feel-sim PRIVATE feel_controller)
add_executable(feel-tests test/controller_tests.cpp)
target_link_libraries(feel-tests PRIVATE feel_controller)
enable_testing()
add_test(NAME firmware_controller COMMAND feel-tests)
